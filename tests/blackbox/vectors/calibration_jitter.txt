# Calibration jitter rejection
# When adjusted_input > 1500 but differs from last_input by > 50,
# enter_calibration_count resets to 0.

@config
armed=0
inputSet=1
servoPwm=1
dshot=0

@sequence
# Set up calibration entry state with jitter
config adjusted_input=1600
config last_input=1500
config enter_calibration_count=45

# transfercomplete with getAbsDif(1600, 1500) = 100 > 50 → reset count
config dma_0=1000 dma_1=2500
tick 1 | transfer=1 |

# Stable input — count should increment
config adjusted_input=1600
config last_input=1600
config enter_calibration_count=45
tick 1 | transfer=1 |
config transfer=0
