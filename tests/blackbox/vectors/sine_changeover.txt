# Sine-to-BLDC changeover
# With input > 200, phase_A_position is forced to 0 which triggers
# the changeover: stepper_sine=0, running=1, old_routine=1.

@tuning am32

@config
armed=1
inputSet=1
dshot=1
EDT_ARMED=1
stepper_sine=1
eeprom.use_sine_start=1
eeprom.comp_pwm=1
eeprom.motor_poles=14
eeprom.stall_protection=1

@sequence
# High throttle (> 200) forces phase_A_position=0 → changeover. The stepper
# reads adjusted_input (AM32 `input`, post sine-start mapping), which the
# input stage publishes AFTER the stepper runs in each pass (firmware main
# loop and harness alike), so the changeover lands one pass later.
tick 1 | throttle=500 |
tick 1 | throttle=500 | stepper_sine=0 running=1 old_routine=1 step=5 commutation_interval=4.5ms
