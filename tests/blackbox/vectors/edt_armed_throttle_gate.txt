# DShot throttle is independent from EDT armed state.
# EDT telemetry/arm mode is an extension; normal DShot throttle must still
# update newinput when EDT_ARMED=0.

@config
armed=0
inputSet=1
dshot=1
EDT_ARM_ENABLE=0

@sequence
# Normal DShot: throttle is accepted without EDT arming.
config EDT_ARMED=0
tick 1  | dshot_frame=1500 | newinput=1500

# Normal DShot: throttle remains accepted if EDT_ARMED happens to be set.
config EDT_ARMED=1
tick 1  | dshot_frame=1500 | newinput=1500

# EDT-arm mode: non-zero throttle is blocked until EDT_ARMED is set.
config EDT_ARM_ENABLE=1
config EDT_ARMED=0
# Zero throttle clears the prior accepted input before checking the gated path.
tick 1  | dshot_frame=0    | newinput=0
tick 1  | dshot_frame=1500 | newinput=0

config EDT_ARMED=1
tick 1  | dshot_frame=1500 | newinput=1500

# Zero throttle remains accepted.
config EDT_ARMED=0
tick 1  | dshot_frame=0    | newinput=0
