# EDT-arm zero-throttle disarm
# When EDT_ARM_ENABLE=1, a zero-throttle DShot frame clears EDT_ARMED.

@config
armed=1
inputSet=1
dshot=1
EDT_ARMED=1
EDT_ARM_ENABLE=1

@sequence
config EDT_ARM_ENABLE=0
config eeprom.input_type=4
load_eeprom
tick 1  | | edt_arm_enable=1

# Zero throttle with EDT_ARM_ENABLE clears EDT_ARMED.
tick 1  | dshot_frame=0    | newinput=0 edt_armed=0 edt_arm_enable=1

# Subsequent non-zero throttle is blocked until EDT arming succeeds again.
tick 1  | dshot_frame=1500 | newinput=0 edt_armed=0 edt_arm_enable=1
