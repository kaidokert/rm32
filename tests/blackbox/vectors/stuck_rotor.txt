# Stuck rotor detection via interval timer timeout
# Covers: REQ-MOTOR-STOP_DETECTION, REQ-PROT-STUCK_ROTOR
#
# When bemf_timeout_happened exceeds the timeout threshold,
# stuck_rotor_protection latches input=0 and cuts phases in ISR context.
#
# The separate stuck_rotor_rate vector covers interval timer reset behavior.

@config
armed=1
inputSet=1
dshot=1
EDT_ARMED=1
running=1
old_routine=1
eeprom.stuck_rotor_protection=1

@sequence
# Normal: low interval timer, motor runs
tick 1  | throttle=500 interval_timer=1000 | running=1 input=500

# Drive repeated high interval samples, then force the latch threshold.
tick 1  | throttle=500 interval_timer=50000 |
tick 1  | throttle=500 interval_timer=50000 |
tick 1  | throttle=500 interval_timer=50000 |
tick 1  | throttle=500 interval_timer=50000 |
tick 1  | throttle=500 interval_timer=50000 |
tick 1  | throttle=500 interval_timer=50000 |
tick 1  | throttle=500 interval_timer=50000 |
tick 1  | throttle=500 interval_timer=50000 |
tick 1  | throttle=500 interval_timer=50000 |
tick 1  | throttle=500 interval_timer=50000 |
tick 1  | throttle=500 interval_timer=50000 |
tick 1  | throttle=500 interval_timer=50000 |
tick 1  | throttle=500 interval_timer=50000 |
tick 1  | throttle=500 interval_timer=50000 |
tick 1  | throttle=500 interval_timer=50000 |
tick 1  | throttle=500 interval_timer=50000 |
tick 1  | throttle=500 interval_timer=50000 |
tick 1  | throttle=500 interval_timer=50000 |
tick 1  | throttle=500 interval_timer=50000 |
tick 1  | throttle=500 interval_timer=50000 |

# After latch: input is cut and phases are turned off in ISR context.
tick 1  | throttle=500 bemf_timeout_happened=11 | input=0 alloff_count>0 mask_interrupts_count>0
